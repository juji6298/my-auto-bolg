import os
import requests
from requests.auth import HTTPBasicAuth
import google.generativeai as genai

# 1. 제미나이 API 키 설정
api_key = os.environ.get("GEMINI_API_KEY")
if not api_key:
    print("경고: GEMINI_API_KEY 환경 변수가 설정되지 않았습니다.")
else:
    genai.configure(api_key=api_key)

def fetch_government_subsidies():
    """1단계: 정부 지원금 데이터 수집"""
    url = "http://apis.data.go.kr/B554287/LocalGovernmentSubsidies/getSubsidyList"
    service_key = os.environ.get("PUBLIC_DATA_API_KEY")
    
    params = {
        'serviceKey': service_key,
        'pageNo': '1',
        'numOfRows': '3',
        'dataType': 'JSON'
    }
    
    try:
        if not service_key or service_key == "YOUR_PUBLIC_DATA_API_KEY_HERE":
            raise ValueError("공공데이터 API 키가 입력되지 않았습니다. 테스트 모드로 전환합니다.")

        response = requests.get(url, params=params, timeout=10)
        response.raise_for_status()
        data = response.json()
        
        items = data.get('response', {}).get('body', {}).get('items', {}).get('item', [])
        
        parsed_posts = []
        for item in items:
            title = item.get('servNm', '지원금 제목 없음')
            target = item.get('trgterIndvdlDscn', '지원 대상 정보 없음')
            content = item.get('servDgst', '상세 내용 없음')
            
            parsed_posts.append({
                "title": title,
                "summary": f"지원 대상: {target}\n\n요약: {content}"
            })
            
        if not parsed_posts:
            raise ValueError("수집된 데이터가 없습니다.")
            
        return parsed_posts

    except Exception as e:
        print(f"데이터 수집 안내: {e} (샘플 테스트 데이터를 사용합니다)")
        return [{
            "title": "2026 청년 도약 및 주거 지원금 총정리",
            "summary": "지원 대상: 무주택 청년 및 중소기업 재직자, 요약: 정부에서 지원하는 주거비 및 자산 형성 보조금 제도"
        }]

def generate_blog_post_with_gemini(raw_title, raw_summary):
    """2단계: 제미나이(Gemini)로 블로그 포스팅 윤문"""
    try:
        model = genai.GenerativeModel('gemini-2.5-flash')
        
        prompt = f"""
        너는 전문적이고 흡입력 있는 재테크/정책 블로거야. 
        아래의 정부 지원금 원본 데이터를 바탕으로, 블로그 독자들이 읽기 쉽고 흥미를 느낄 수 있는 블로그 포스팅 본문을 작성해줘.

        [원본 데이터]
        - 지원금 명칭: {raw_title}
        - 상세 내용: {raw_summary}

        [작성 가이드라인]
        1. 첫 번째 줄에는 반드시 블로그 제목만 적어줘.
        2. 두 번째 줄부터는 본문 내용을 작성해줘. 본문은 도입부, 지원 대상, 신청 방법, 꿀팁 순으로 가독성 좋게 작성해줘.
        3. 말투는 친근하고 전문적인 '~해요, ~랍니다' 체를 사용해줘.
        """

        response = model.generate_content(prompt)
        return response.text
    except Exception as e:
        print(f"제미나이 글 생성 중 에러 발생: {e}")
        return None

def publish_to_wordpress(title, content):
    """3단계: 워드프레스에 자동 발행 (안전을 위해 처음엔 임시저장 'draft')"""
    wp_url = os.environ.get("WP_URL")
    wp_username = os.environ.get("WP_USERNAME")
    wp_app_password = os.environ.get("WP_APP_PASSWORD")
    
    if not all([wp_url, wp_username, wp_app_password]):
        print("워드프레스 연동 정보가 부족합니다. 로컬 테스트 출력을 진행합니다.")
        print(f"\n[생성된 제목]: {title}\n[생성된 본문]:\n{content}\n")
        return

    post_data = {
        'title': title,
        'content': content,
        'status': 'draft'
    }
    
    auth = HTTPBasicAuth(wp_username, wp_app_password)
    
    try:
        response = requests.post(wp_url, json=post_data, auth=auth, timeout=10)
        if response.status_code == 201:
            print(f"성공! [ {title} ] 글이 워드프레스 임시저장함에 쏙 들어갔습니다.")
        else:
            print(f"발행 실패 (에러 코드: {response.status_code})")
            print(response.text)
    except Exception as e:
        print(f"워드프레스 통신 중 에러 발생: {e}")

if __name__ == "__main__":
    print("🤖 제미나이 기반 정부지원금 자동화 봇을 가동합니다...")
    
    subsidies = fetch_government_subsidies()
    
    for item in subsidies:
        raw_title = item['title']
        raw_summary = item['summary']
        
        print(f"\n처리 중인 항목: {raw_title}")
        
        generated_content = generate_blog_post_with_gemini(raw_title, raw_summary)
        
        if generated_content:
            lines = generated_content.strip().split('\n')
            post_title = lines[0].replace('#', '').replace('제목:', '').strip() if lines else raw_title
            post_content = "\n".join(lines[1:]).strip() if len(lines) > 1 else generated_content
            
            publish_to_wordpress(post_title, post_content)
