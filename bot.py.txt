import os
import google.generativeai as genai

# 제미나이 API 키 설정
api_key = os.environ.get("GEMINI_API_KEY")
if not api_key:
    print("경고: GEMINI_API_KEY 환경 변수가 설정되지 않았습니다.")
else:
    genai.configure(api_key=api_key)

def generate_blog_post_with_gemini():
    """제미나이(Gemini)로 티스토리에 올릴 알짜배기 정부 지원금 글 생성"""
    try:
        model = genai.GenerativeModel('gemini-2.5-flash')
        
        prompt = """
        너는 전문적이고 흡입력 있는 대한민국 재테크/정책 블로거야. 
        청년과 직장인들이 눈을 뗄 수 없는 정부 지원금(예: 청년도약계좌, 주거 지원금 등) 중 하나를 골라서, 티스토리 블로그에 바로 올릴 수 있는 완성도 높은 포스팅 본문을 작성해줘.

        [작성 가이드라인]
        1. 첫 번째 줄에는 독자의 시선을 사로잡는 매력적인 블로그 제목만 적어줘.
        2. 두 번째 줄부터는 본문을 작성해줘. 도입부, 지원 대상, 신청 방법, 실전 꿀팁 순으로 가독성 좋게 마크다운 형식으로 작성해줘.
        3. 말투는 친근하고 전문적인 '~해요, ~랍니다' 체를 사용해줘.
        """

        response = model.generate_content(prompt)
        return response.text
    except Exception as e:
        print(f"제미나이 글 생성 중 에러 발생: {e}")
        return None

if __name__ == "__main__":
    print("🤖 [티스토리 맞춤형] 제미나이가 알짜배기 지원금 글을 작성합니다...")
    
    generated_content = generate_blog_post_with_gemini()
    
    if generated_content:
        print("\n" + "="*50)
        print("✨ [생성된 블로그 포스팅 미리보기]")
        print("="*50 + "\n")
        print(generated_content)
        print("\n" + "="*50)
